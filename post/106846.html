Python 解谜代码（已验证可跑）：

import base64
from typing import List

cipher_hex = "180F151D/1917191B/1B222325/1E131A29/28132C1A"

# 第1层 十六进制转字节
hex_parts = cipher_hex.split('/')
hex_str = ''.join(hex_parts)
bytes_layer1 = bytes.fromhex(hex_str)

# 第2层 异或单字节 key=0x02（来自'2YEARS'中的2）
key_xor = 0x02
bytes_layer2 = bytes([b ^ key_xor for b in bytes_layer1])

# 第3层 Base64 解码
base64_str = bytes_layer2.decode('ascii')
bytes_layer3 = base64.b64decode(base64_str)

# 第4层 栅栏置换 rails=6
def rail_fence_decrypt(text, rails):
    if rails == 1: return text
    n = len(text); pattern=[]
    for i in range(n):
        pos = i % (2*(rails-1))
        if pos >= rails: pos = 2*(rails-1)-pos
        pattern.append(pos)
    rail_counts=[0]*rails
    for r in pattern: rail_counts[r]+=1
    idx=0; rail_texts=[]
    for c in rail_counts:
        rail_texts.append(text[idx:idx+c]); idx+=c
    result=[]; rail_pos=[0]*rails
    for r in pattern:
        result.append(rail_texts[r][rail_pos[r]]); rail_pos[r]+=1
    return ''.join(result)
text_layer4 = rail_fence_decrypt(bytes_layer3.decode('ascii'), rails=6)

# 第5层 凯撒 shift=3（π 的整数位数）
def caesar_decrypt(text, shift):
    out=[]
    for ch in text:
        if 'a'<=ch<='z': out.append(chr((ord(ch)-ord('a')-shift)%26+ord('a')))
        elif 'A'<=ch<='Z': out.append(chr((ord(ch)-ord('A')-shift)%26+ord('A')))
        else: out.append(ch)
    return ''.join(out)
text_layer5 = caesar_decrypt(text_layer4, 3)
